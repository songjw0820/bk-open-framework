cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bk_ota C)
endif()

if(NOT DEFINED CONFIG_BK_OTA)
    set(CONFIG_BK_OTA OFF)
endif()

if(CONFIG_BK_OTA)
    set(BK_OTA_SOURCES src/bk_ota.c)
    if(COMMAND armino_component_register)
        armino_component_register(SRCS ${BK_OTA_SOURCES} INCLUDE_DIRS include REQUIRES bk_core)
    else()
        add_library(bk_ota STATIC ${BK_OTA_SOURCES})
        target_include_directories(bk_ota PUBLIC include)
        target_include_directories(bk_ota PRIVATE ../bk_core/include)
        target_compile_features(bk_ota PUBLIC c_std_11)
    endif()

    option(BK_OTA_BUILD_TESTS "Build bk_ota host tests" OFF)
    if(BK_OTA_BUILD_TESTS AND NOT COMMAND armino_component_register)
        enable_testing()
        add_executable(bk_ota_test tests/test_ota_state.c ${BK_OTA_SOURCES})
        target_include_directories(bk_ota_test PRIVATE include ../bk_core/include)
        target_compile_features(bk_ota_test PRIVATE c_std_11)
        add_test(NAME bk_ota_test COMMAND bk_ota_test)
    endif()
elseif(COMMAND armino_component_register)
    armino_component_register()
endif()
