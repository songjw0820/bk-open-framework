cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bk_ui C)
endif()

if(NOT DEFINED CONFIG_BK_UI)
    set(CONFIG_BK_UI OFF)
endif()

if(CONFIG_BK_UI)
    set(BK_UI_SOURCES src/bk_ui.c)
    if(COMMAND armino_component_register)
        armino_component_register(SRCS ${BK_UI_SOURCES} INCLUDE_DIRS include REQUIRES bk_core bk_board)
    else()
        add_library(bk_ui STATIC ${BK_UI_SOURCES})
        target_include_directories(bk_ui PUBLIC include)
        target_include_directories(bk_ui PRIVATE ../bk_core/include ../bk_board/include)
        target_compile_features(bk_ui PUBLIC c_std_11)
    endif()

    option(BK_UI_BUILD_TESTS "Build bk_ui host tests" OFF)
    if(BK_UI_BUILD_TESTS AND NOT COMMAND armino_component_register)
        enable_testing()
        add_executable(bk_ui_test tests/test_ui.c ${BK_UI_SOURCES})
        target_include_directories(bk_ui_test PRIVATE include ../bk_core/include ../bk_board/include)
        target_compile_features(bk_ui_test PRIVATE c_std_11)
        add_test(NAME bk_ui_test COMMAND bk_ui_test)
    endif()
elseif(COMMAND armino_component_register)
    armino_component_register()
endif()
