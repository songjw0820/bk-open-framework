cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bk_net C)
endif()

set(BK_NET_SOURCES src/bk_net.c src/bk_provision.c src/bk_lan.c)
if(COMMAND armino_component_register)
    armino_component_register(SRCS ${BK_NET_SOURCES} INCLUDE_DIRS include REQUIRES bk_core)
else()
    add_library(bk_net STATIC ${BK_NET_SOURCES})
    target_include_directories(bk_net PUBLIC include)
    target_include_directories(bk_net PRIVATE ../bk_core/include)
    target_compile_features(bk_net PUBLIC c_std_11)
endif()

option(BK_NET_BUILD_TESTS "Build bk_net host tests" OFF)
if(BK_NET_BUILD_TESTS)
    enable_testing()
    add_executable(bk_net_test tests/test_net_state.c ${BK_NET_SOURCES})
    target_include_directories(bk_net_test PRIVATE include ../bk_core/include)
    target_compile_features(bk_net_test PRIVATE c_std_11)
    add_test(NAME bk_net_test COMMAND bk_net_test)
endif()
