cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bk_media C)
endif()

if(NOT DEFINED CONFIG_BK_MEDIA)
    set(CONFIG_BK_MEDIA OFF)
endif()

if(CONFIG_BK_MEDIA)
    set(BK_MEDIA_SOURCES src/bk_media.c)
    if(COMMAND armino_component_register)
        armino_component_register(SRCS ${BK_MEDIA_SOURCES} INCLUDE_DIRS include REQUIRES bk_core bk_board)
    else()
        add_library(bk_media STATIC ${BK_MEDIA_SOURCES})
        target_include_directories(bk_media PUBLIC include)
        target_include_directories(bk_media PRIVATE ../bk_core/include ../bk_board/include)
        target_compile_features(bk_media PUBLIC c_std_11)
    endif()

    option(BK_MEDIA_BUILD_TESTS "Build bk_media host tests" OFF)
    if(BK_MEDIA_BUILD_TESTS AND NOT COMMAND armino_component_register)
        enable_testing()
        add_executable(bk_media_test tests/test_media.c ${BK_MEDIA_SOURCES})
        target_include_directories(bk_media_test PRIVATE include ../bk_core/include ../bk_board/include)
        target_compile_features(bk_media_test PRIVATE c_std_11)
        add_test(NAME bk_media_test COMMAND bk_media_test)
    endif()
endif()
